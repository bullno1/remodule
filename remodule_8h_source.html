427static char*
428remodule_find_pdb_ref(char* image, size_t image_size, size_t* capacity) {
429 IMAGE_DOS_HEADER* dos_header = remodule_image_at(image, image_size, 0, sizeof(IMAGE_DOS_HEADER));
430 if (
431 dos_header == NULL
432 || dos_header->e_magic != IMAGE_DOS_SIGNATURE
433 || dos_header->e_lfanew < 0
434 ) {
435 return NULL;
436 }
437
438 // The module is loaded into this process so it has the same header layout
439 size_t nt_headers_offset = (size_t)dos_header->e_lfanew;
440 IMAGE_NT_HEADERS* nt_headers = remodule_image_at(image, image_size, nt_headers_offset, sizeof(IMAGE_NT_HEADERS));
441 if (
442 nt_headers == NULL
443 || nt_headers->Signature != IMAGE_NT_SIGNATURE
444 || nt_headers->OptionalHeader.Magic != IMAGE_NT_OPTIONAL_HDR_MAGIC
445 || nt_headers->OptionalHeader.NumberOfRvaAndSizes <= IMAGE_DIRECTORY_ENTRY_DEBUG
446 ) {
447 return NULL;
448 }
449
450 // The debug directory is located by its virtual address
451 IMAGE_DATA_DIRECTORY debug_dir = nt_headers->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_DEBUG];
452 size_t num_sections = nt_headers->FileHeader.NumberOfSections;
453 IMAGE_SECTION_HEADER* sections = remodule_image_at(
454 image, image_size,
455 nt_headers_offset
456 + offsetof(IMAGE_NT_HEADERS, OptionalHeader)
457 + nt_headers->FileHeader.SizeOfOptionalHeader,
458 num_sections * sizeof(IMAGE_SECTION_HEADER)
459 );
460 if (sections == NULL) { return NULL; }
461
462 IMAGE_DEBUG_DIRECTORY* entries = NULL;
463 for (size_t i = 0; i < num_sections; ++i) {
464 if (
465 debug_dir.VirtualAddress >= sections[i].VirtualAddress
466 && debug_dir.VirtualAddress - sections[i].VirtualAddress < sections[i].SizeOfRawData
467 ) {
468 entries = remodule_image_at(
469 image, image_size,
470 (size_t)sections[i].PointerToRawData + (debug_dir.VirtualAddress - sections[i].VirtualAddress),
471 debug_dir.Size
472 );
473 break;
474 }
475 }
476 if (entries == NULL) { return NULL; }
477
478 size_t num_entries = debug_dir.Size / sizeof(IMAGE_DEBUG_DIRECTORY);
479 for (size_t i = 0; i < num_entries; ++i) {
480 if (
481 entries[i].Type != IMAGE_DEBUG_TYPE_CODEVIEW
482 || entries[i].SizeOfData <= sizeof(remodule_pdb_ref_t)
483 ) {
484 continue;
485 }
486
487 remodule_pdb_ref_t* ref = remodule_image_at(image, image_size, entries[i].PointerToRawData, entries[i].SizeOfData);
488 size_t path_capacity = entries[i].SizeOfData - sizeof(remodule_pdb_ref_t);
489 if (
490 ref != NULL
491 && ref->signature == 0x53445352 // RSDS
492 && memchr(ref->path, '\0', path_capacity) != NULL
493 ) {
494 *capacity = path_capacity;
495 return ref->path;
496 }
497 }
498
499 return NULL;
500}

502// A debugger locks the PDB of a loaded module, even after the module is
503// unloaded, so the linker can not write to it.
504// Give the temporary module its own copy of the PDB and leave the original
505// alone.
506// When this is not possible, the module is left as is.
507static void
508remodule_dynlib_redirect_pdb(const char* module_path, const char* dir, char* pdb_path) {
509 pdb_path[0] = '\0';
510
511 HANDLE file = CreateFileA(
512 module_path,
513 GENERIC_READ | GENERIC_WRITE,
514 0,
515 NULL,
516 OPEN_EXISTING,
517 FILE_ATTRIBUTE_NORMAL,
518 NULL
519 );
520 if (file == INVALID_HANDLE_VALUE) { return; }
521
522 LARGE_INTEGER file_size = { 0 };
523 GetFileSizeEx(file, &file_size);
524 HANDLE mapping = CreateFileMappingA(file, NULL, PAGE_READWRITE, 0, 0, NULL);
525 char* image = mapping != NULL ? MapViewOfFile(mapping, FILE_MAP_WRITE, 0, 0, 0) : NULL;
526
527 size_t ref_capacity = 0;
528 char* ref = image != NULL
529 ? remodule_find_pdb_ref(image, (size_t)file_size.QuadPart, &ref_capacity)
530 : NULL;
531 if (ref != NULL) {
532 // The PDB might have been moved along with the module
533 char moved_path[MAX_PATH];
534 const char* original_path = ref;
535 if (
536 GetFileAttributesA(ref) == INVALID_FILE_ATTRIBUTES
537 && snprintf(moved_path, sizeof(moved_path), "%s%s", dir, remodule_file_part(ref)) < (int)sizeof(moved_path)
538 ) {
539 original_path = moved_path;
540 }
541
542 // The copy is made next to the original.
543 // Its name is made by writing a number over the end of the original
544 // name, extension included.
545 // This keeps the length so the new path always fits in the old record.
546 size_t original_path_len = strlen(original_path);
547 size_t name_len = strlen(remodule_file_part(original_path));
548 int max_copies = 1;
549 for (size_t i = 0; i < name_len && i < 3; ++i) { max_copies *= 10; }
550
551 // The copies made for the previous loads might still be held by a
552 // debugger so every load needs a new name
553 bool copied = false;
554 for (int i = 0; i < max_copies && name_len > 0 && original_path_len < MAX_PATH; ++i) {
555 memcpy(pdb_path, original_path, original_path_len + 1);
556 char* digit = pdb_path + original_path_len;
557 int number = i;
558 do {
559 *(--digit) = (char)('0' + number % 10);
560 number /= 10;
561 } while (number > 0);
562
563 copied = CopyFileA(original_path, pdb_path, TRUE) != FALSE;
564 if (copied) { break; }
565
566 DWORD error = GetLastError();
567 if (error != ERROR_FILE_EXISTS && error != ERROR_ALREADY_EXISTS) { break; }
568 }
569
570 // When the PDB was moved, the new path might not fit in the old record.
571 // A debugger also looks for the PDB next to the module so the file
572 // name alone is enough.
573 const char* new_ref = pdb_path;
574 if (strlen(new_ref) >= ref_capacity) {
575 new_ref = remodule_file_part(pdb_path);
576 }
577
578 if (copied && strlen(new_ref) < ref_capacity) {
579 size_t new_ref_len = strlen(new_ref);
580 memcpy(ref, new_ref, new_ref_len);
581 memset(ref + new_ref_len, 0, ref_capacity - new_ref_len);
582 } else {
583 if (copied) { DeleteFileA(pdb_path); }
584 pdb_path[0] = '\0';
585 }
586 }
587
588 if (image != NULL) { UnmapViewOfFile(image); }
589 if (mapping != NULL) { CloseHandle(mapping); }
590 CloseHandle(file);
591}

593static remodule_dynlib_t
594remodule_dynlib_open(const char* path) {
595 // Load once to find the real absolute path
596 HMODULE module = LoadLibraryExA(
597 path,
598 NULL,
599 DONT_RESOLVE_DLL_REFERENCES
600 );
601 if (module == NULL) { return NULL; }
602
603 char watch_path[MAX_PATH];
604 DWORD watch_path_len = GetModuleFileNameA(module, watch_path, sizeof(watch_path));
605 REMODULE_ASSERT(watch_path_len > 0, "Could not get module name");
606 FreeLibrary(module);
607
608 // Create a temporary file name
609 char dir_buf[MAX_PATH];
610 char name_buf[MAX_PATH];
611 char tmp_name_buf[MAX_PATH];
612 char* file_part;
613
614 GetFullPathNameA(watch_path, sizeof(dir_buf), dir_buf, &file_part);
615 size_t name_len = strlen(file_part);
616 memcpy(name_buf, file_part, name_len);
617 *file_part = '\0';
618 GetTempFileNameA(dir_buf, name_buf, 0, tmp_name_buf);
619
620 // Copy the file over
621 REMODULE_ASSERT(CopyFileA(watch_path, tmp_name_buf, FALSE), "Could not create temporary file");
622
623 char pdb_path[MAX_PATH];
624 remodule_dynlib_redirect_pdb(tmp_name_buf, dir_buf, pdb_path);
625
626 // Load the temporary file
627 module = LoadLibraryA(tmp_name_buf);
628 if (module == NULL) {
629 DWORD error = GetLastError();
630 DeleteFileA(tmp_name_buf);
631 if (pdb_path[0] != '\0') { DeleteFileA(pdb_path); }
632 SetLastError(error);
633 return NULL;
634 }
635
636 remodule_dynlib_t lib = malloc(sizeof(remodule_dynlib_info_t) + watch_path_len + 1);
637 lib->handle = module;
638 memcpy(lib->pdb_path, pdb_path, sizeof(pdb_path));
639 memcpy(lib->watch_path, watch_path, watch_path_len);
640 lib->watch_path[watch_path_len] = '\0';
641 return lib;
642}

789void
790remodule_reload(remodule_t* mod) {
791 mod->info.entry(REMODULE_OP_BEFORE_RELOAD, mod->userdata);
792
793 // Store all static vars in a host-allocated buffer
794 int num_vars = 0;
795 size_t val_buffer_size = 0;
796 size_t name_buffer_size = 0;
797 for (
798 const remodule_var_info_t* const* itr = mod->info.var_info_begin;
799 itr != mod->info.var_info_end;
800 ++itr
801 ) {
802 if (*itr == NULL) { continue; }
803 remodule_var_info_t var_info = **itr;
804
805 ++num_vars;
806 val_buffer_size += var_info.value_size;
807 name_buffer_size += var_info.name_length;
808 }
809
810 void* tmp_buf = malloc(
811 num_vars * sizeof(remodule_tmp_var_storage_t)
812 + val_buffer_size
813 + name_buffer_size
814 );
815 remodule_tmp_var_storage_t* entry_ptr = tmp_buf;
816 char* data_ptr = (char*)(entry_ptr + num_vars);
817
818 for (
819 const remodule_var_info_t* const* itr = mod->info.var_info_begin;
820 itr != mod->info.var_info_end;
821 ++itr
822 ) {
823 if (*itr == NULL) { continue; }
824 remodule_var_info_t var_info = **itr;
825
826 remodule_tmp_var_storage_t* entry = entry_ptr++;
827

853 ++var_itr

866 ) {
867 memcpy(var_info.value_addr, storage->value, storage->value_size);
